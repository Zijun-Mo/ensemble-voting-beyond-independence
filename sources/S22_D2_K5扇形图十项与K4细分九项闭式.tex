\section{问题定义与证书对偶}
设 $V=\{0,1,2,3,4\}$，$X_i\in\{0,1\}$ 为第 $i$ 个模型的错误指示变量。已知
\[
  \mathbb E X_i=q_i,\qquad
  \mathbb E(X_iX_j)\le b_{ij},
  \qquad 0<q_i<\frac12.
\]
记 $\calP(q,b)$ 为满足约束的非空联合分布多面体，$F:\{0,1\}^5\to\{0,1\}$ 为确定性 self-dual 规则：$F(x)+F(\boldsymbol1-x)=1$。定义
\[
 \Rd(q,b)=\min_{F\text{ self-dual}}\max_{\pi\in\calP(q,b)}\mathbb E_\pi F(X).
\]
高精度区域定义为
\begin{equation}\label{eq:R}
\calR=\left\{(q,b):\sum_{i=0}^4q_i-3\min_iq_i\le1,\;\calP(q,b)\ne\varnothing\right\}.
\end{equation}
对任何满足 $b_{ij}=\min(q_i,q_j)$ 的边，其共错上界为真空约束，可直接忽略。下文将非真空边限制在给定的有效边集 $E$ 上；其余边不施加额外信息。

在固定 $E$ 上的二次对偶证书写作
\begin{equation}\label{eq:Q}
 Q(x)=a+\sum_i\beta_i x_i+\sum_{ij\in E}\gamma_{ij}x_ix_j,
 \qquad \gamma_{ij}\ge0,
\end{equation}
目标函数为
\begin{equation}\label{eq:L}
 L_Q(q,b)=a+\sum_i\beta_iq_i+\sum_{ij\in E}\gamma_{ij}b_{ij}.
\end{equation}
若 $Q(x)\ge F(x)$，则对任意可行联合分布均有 $\mathbb EF\le L_Q$。

设\emph{混合对偶}多面体
\begin{equation}\label{eq:mixed}
\mathscr D_E=\left\{Q:\ Q(x)\ge0,\;Q(x)+Q(\boldsymbol1-x)\ge1\;(\forall x),\;
\gamma_e\ge0\;(e\in E)\right\}.
\end{equation}
这一多面体只有实系数，\textbf{不}预设其为整数，也\textbf{不}声称混合 minimax 等于确定性 minimax。合法确定性证书另外满足
\begin{equation}\label{eq:orient}
 \max\{Q(x),Q(\boldsymbol1-x)\}\ge1\quad(\forall x).
\end{equation}
固定任何一对 $x_*,\boldsymbol1-x_*$，式 \eqref{eq:orient} 至少意味着
\[
 Q(x_*)\ge1\quad\text{或}\quad Q(\boldsymbol1-x_*)\ge1.
\]
这是下文有限完整性证明的关键。

\section{扇形图的十项精确闭式}
取
\begin{equation}\label{eq:fan}
 E_{\rm fan}=\{01,02,03,04,12,14,23\}.
\end{equation}
定义四条奇环费用
\begin{align*}
 C_{012}&=b_{01}+b_{02}+b_{12},&
 C_{014}&=b_{01}+b_{04}+b_{14},\\
 C_{023}&=b_{02}+b_{03}+b_{23},&
 C_{5,\mathrm{fan}}&=b_{03}+b_{04}+b_{12}+b_{14}+b_{23}.
\end{align*}
再定义扇形正截距项
\begin{equation}\label{eq:Bfan}
 B_{\mathrm{fan}}=1-q_0-q_1-q_2+
 b_{01}+b_{02}+b_{03}+b_{04}+b_{12}+b_{14}+b_{23}.
\end{equation}

\begin{theorem}[扇形图十项定理]\label{thm:fan}
若所有严格有效共错边均包含在 $E_{\rm fan}$，参数可行且满足 \eqref{eq:R}，则
\begin{equation}\label{eq:fan_closed}
\boxed{\Rd=\min\{q_0,q_1,q_2,q_3,q_4,C_{012},C_{014},C_{023},C_{5,\mathrm{fan}},B_{\mathrm{fan}}\}.}
\end{equation}
这是整个 $\calR$ 上的精确等式，而非只在附加三角形边际条件下成立。
\end{theorem}

\paragraph{显式上界。}
前九项分别由单一模型独裁与奇环多数投票实现。最后一项来自
\begin{equation}\label{eq:Qfan}
 Q_{\rm fan}(x)=1-x_0-x_1-x_2+\sum_{ij\in E_{\rm fan}}x_ix_j.
\end{equation}
令 $N=x_0+x_1+x_2$，则
\[
 Q_{\rm fan}(x)=\underbrace{1-N+\binom N2}_{\mathbf1\{x_0=x_1=x_2\}}
 +x_0x_3+x_0x_4+x_1x_4+x_2x_3.
\]
因此 $Q\ge0$。若 $x_0,x_1,x_2$ 不全相同，则外部四条边组成连接这三点的路径 $1-4-0-3-2$；其上总有同色边，故在互补两侧中至少一侧使 $Q\ge1$。这样 $Q$ 满足式 \eqref{eq:orient}，其导出的 self-dual 加权投票权重与 $(2,1,1,2,2)$ 成比例，并可在平局时采取符合 $Q$ 的 self-dual 裁决。

\paragraph{完整性认证。}
对该七边图，$\mathscr D_{E_{\rm fan}}$ 的齐次化锥经精确双描述认证，具有 $42$ 个非冗余有限点生成元（验证数据存储 $43$ 个，其中一个冗余）、$28$ 条衰退射线与 $49$ 个非冗余面。固定互补对（二进制状态编号 $7$ 和 $24$），两个取向分支各产生 $239$ 个\emph{覆盖全部分支顶点的候选点}；$478$ 份有理数 Farkas 证书在整个可行区域 $\calR$ 上逐项证明其目标不低于式 \eqref{eq:fan_closed}。可行分布使所有非负衰退方向在目标中非负，由此得到精确下界。

\begin{remark}[不能误用随机化等价]
即使严格位于 $\calR$ 内，混合 self-dual 的 minimax 值也可能低于确定性值。扇形图存在有理参数使确定性值为 $11/1600$，而混合值为 $3/640$；因此完整性证明\emph{必须}使用确定性取向约束，不能仅凭混合多面体的有限极点表宣称等式。
\end{remark}

\section{$K_4$ 单边细分图的九项完整闭式：本轮新增}
取
\begin{equation}\label{eq:subdiv}
 E_{\rm sub}=\{02,03,04,12,13,14,24\}.
\end{equation}
其结构可视为 $K_4$ 的边 $01$ 被路径 $0-3-1$ 代替，因此\emph{含 $K_4$ minor}。定义两条三角形与两条五环的费用
\begin{align}\label{eq:cycles}
 T_0&=b_{02}+b_{04}+b_{24},&T_1&=b_{12}+b_{14}+b_{24},\\
 P_0&=b_{02}+b_{03}+b_{13}+b_{14}+b_{24},&
 P_1&=b_{03}+b_{04}+b_{12}+b_{13}+b_{24}.
\end{align}

\begin{theorem}[$K_4$ 单边细分图九项定理]\label{thm:subdiv}
若所有严格有效共错边均包含在 $E_{\rm sub}$，参数可行且满足 \eqref{eq:R}，则
\begin{equation}\label{eq:sub_closed}
\boxed{\Rd=\min\{q_0,q_1,q_2,q_3,q_4,T_0,T_1,P_0,P_1\}.}
\end{equation}
特别地，尽管该图含 $K_4$ minor，九项均由独裁或奇环多数投票实现，无需增加正截距二阶矩项。
\end{theorem}

\paragraph{显式上界。}
对任何奇环 $C$，奇环多数失败意味着至少两个相邻模型同时出错，因而
\[
 \mathbb P(\text{$C$-majority fails})\le\sum_{ij\in C}b_{ij}.
\]
五个独裁规则与四条奇环给出式 \eqref{eq:sub_closed} 的上界。

\paragraph{全部实系数证书的有限覆盖。}
对七条边，式 \eqref{eq:mixed} 的 $13$ 个系数满足 $32$ 条非负性约束、$16$ 条互补求和约束及 $7$ 条 $\gamma\ge0$ 约束。引入齐次化坐标 $z\ge0$ 后，原始锥由 $56$ 条整数线性不等式给出。本轮精确验证找到如下\emph{完整}生成元表：
\[
 \boxed{58\text{ 个有限顶点}\; +\;28\text{ 条衰退射线}.}
\]
从这些生成元实施整数双描述，得到恰好 $47$ 个非冗余面；每一个面都属于原始 $56$ 条约束。反之，全部 $86$ 个生成元直接满足全部原始约束，所以两个锥完全相同。所有可行实系数混合证书都已被覆盖。

\paragraph{只需一对互补状态。}
取 $x_*=00011$（状态编号 $3$，取最低位为 $x_0$），其补状态编号为 $28$。任意确定性可行证书必在下面两个集合的并集中：
\[
 \mathscr D_{E_{\rm sub}}\cap\{Q(3)\ge1\}
 \quad\cup\quad
 \mathscr D_{E_{\rm sub}}\cap\{Q(28)\ge1\}.
\]
两分支各有至多 $167$ 个必要候选。这些候选由原始有限顶点、顶点--顶点交截及顶点--射线交截\emph{穷尽式生成}，不需要假设任何整数系数上界。

\paragraph{全区域 Farkas 支配证明。}
引入 $32$ 个联合分布质量 $\pi_x$、$7$ 个边预算变量 $b_e$、辅助变量 $\tau$，把所有可行性条件和 $\tau\le$ 九个显式项写为 $Az\le h$，其中 $z\ge0$ 且 $e^\top z=1$（$e$ 对前 $32$ 个分量取 $1$）。对于每个候选 $Q$，令 $c_Q^\top z=\sum_x\pi_x\sum_i\beta_ix_i+\sum_e\gamma_eb_e-\tau$。所附证明数据逐项提供 $y\ge0$、$\lambda\in\mathbb Q$，精确满足
\begin{equation}\label{eq:farkas}
 c_Q+A^\top y-\lambda e\ge0,\qquad
 a_Q+\lambda-h^\top y\ge0.
\end{equation}
故对任意可行 $z$，
\[
 a_Q+c_Q^\top z
 =(a_Q+\lambda-h^\top y)+
 (c_Q+A^\top y-\lambda e)^\top z+
 y^\top(h-Az)\ge0.
\]
即每个候选都满足 $L_Q\ge\tau$，取 $\tau$ 等于九项式的最小值便得 $L_Q\ge U_9$。两个取向分支共 $334$ 份\emph{精确有理数} Farkas 证书均通过检查；生成元与非冗余面也由模素数秩及 Hadamard 上界作严格认证。加上不改善目标的非负衰退方向，即完成定理 \ref{thm:subdiv} 的下界证明。

\section{可复用的单互补对取向法与剩余结构}
\begin{lemma}[单互补对的充分完整性判据]
对任意固定有效图 $E$，设已严格穷尽混合二次对偶多面体 $\mathscr D_E$ 的有限顶点与衰退射线。若存在某一互补对 $x_*,\boldsymbol1-x_*$，使上述两个定向半空间与 $\mathscr D_E$ 的全部交点候选，在指定矩参数域上都被一个合法显式上界 $U(q,b)$ 支配，则确定性 self-dual minimax 恰好为 $U$。
\end{lemma}
\begin{proof}
任意合法确定性 $Q$ 由于式 \eqref{eq:orient} 必落在两个分支之一。多面体与一个半空间相交后的极点，只可能来自原始极点，或原始顶点--顶点线段、顶点--衰退射线穿越新边界的交点；把所有可能的穿越组合都纳入形成有限超集，足以覆盖全部新极点。其余衰退方向在每个可行 $(q,b)$ 下都无法使目标下降。若有限候选都被 $U$ 支配，则所有确定性 $Q$ 亦被 $U$ 支配；再由显式规则实现 $U$，得取等号。
\end{proof}

截至本轮，在沿用此前已认证的 $938/1024$ 个有效图结论基础上，本轮新增了 $30$ 种 $K_4$ 单边细分图的标号形式，得到 $968/1024$ 种图结构上的精确闭式。余下 $56$ 种标号图分属\emph{四种}无标号结构：两种八边图（分别 $30$、$15$ 种标号形式）、九边图（$10$ 种）与完整 $K_5$（$1$ 种）。这是\emph{图结构计数}，不是概率参数的体积比例。

本轮未证明这些四类稠密核心的无优化闭式，因此不对其提出已经完成的结论。单互补对取向法及完整实系数混合对偶的穷尽认证，是下一阶段处理它们的可重用方法。

def primitive(v):
    div=reduce(gcd,(abs(int(t)) for t in v))
    assert div>0
    return tuple(int(t)//div for t in v)


def dot(a,b):return sum(x*y for x,y in zip(a,b))


def rankmod(rows,p):
    if not rows:return 0
    mat=[[int(x)%p for x in row] for row in rows]
    nr=len(mat);nc=len(mat[0]);rank=0
    for c in range(nc):
        k=next((i for i in range(rank,nr) if mat[i][c]),None)
        if k is None:continue
        mat[rank],mat[k]=mat[k],mat[rank]
        inv=pow(mat[rank][c],p-2,p)
        for j in range(rank+1,nr):
            if mat[j][c]:
                f=mat[j][c]*inv%p
                for z in range(c,nc):mat[j][z]=(mat[j][z]-f*mat[rank][z])%p
        rank+=1
        if rank==min(nr,nc):break
    return rank

def rankmod(rows,p):
    if not rows:return 0

    nr=len(mat);nc=len(mat[0]);rank=0

        if k is None:continue
